\definexp{phd}{University of Edinburgh}{Doctoral Researcher}{11/2023 -- present}
\xpdesc{phd}{industry}{%
  Foundational work on the training dynamics of neural networks used for
  high-dimensional model fitting to noisy experimental data: a typical fit
  carries $\sim$600 free parameters constrained by $\sim$5,000 correlated
  measurements, with experimental uncertainties propagated by Monte Carlo
  bootstrap over $\sim$100 replicas. Developed a
  theoretical framework based on the neural tangent kernel that gives a
  quantitative account of how such fits behave, and that tests whether the
  uncertainties they report are statistically faithful
  (\textit{EPJC 86, 722}). Supervised by Prof. Luigi Del Debbio.}

\definexp{nnpdf}{\href{https://nnpdf.mi.infn.it}{NNPDF Collaboration}}{Member and Developer}{since 11/2023}
\xpdesc{nnpdf}{academic}{%
  Contribute to the maintenance and development of the NNPDF fitting codebase, a production
  Python framework used collaboration-wide to determine the parton distribution functions
  of the proton.}
\xpdesc{nnpdf}{industry}{%
  Maintain and extend the production Python codebase used collaboration-wide for
  large-scale model fitting: code review, CI, and release process with citable a
  \href{https://doi.org/10.5281/zenodo.5362228}{DOI}. Extended the framework to
  account for theoretical uncertainties from power corrections through a
  Bayesian treatment of the data (\textit{EPJC 86, 281}), producing a new tool
  included in the forthcoming NNPDF4.1 release. Delivered within a distributed
  team of 20+ international researchers, with regular group meetings, internal
  presentations, and mentoring of new members. Benchmarked model sets from
  several independent groups, ours included, against out of sample measurements
  (\textit{JHEP 07, 067}).}
